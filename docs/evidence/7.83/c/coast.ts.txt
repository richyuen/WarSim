import { readFileSync } from 'node:fs';
import { SIZE_1938 } from '../../../src/sim/scenario1938';
import { Sim } from '../../../src/sim/sim';
import { navOf, seaOf } from '../../../src/sim/world';
import { loadAssets1938 } from '../../../tools/headless/assets';
const s = new Sim({ scenario: '1938', seed: 99, assets: loadAssets1938(SIZE_1938.w) });
s.load(new Uint8Array(readFileSync('.cache/prof83/h0_y4.bin')));
const w = (s as any).world;
s.step(12);
const { w: W, h: H, supply, controller } = w.cells;
const comp = navOf(w).grid.component; const zoneOf = seaOf(w).zoneOf;
let coast = 0, s0 = 0, s0c = 0;
for (let k = 0; k < W * H; k++) {
  if (comp[k] === 0) continue;
  const x = k % W, row = k - x;
  const a = row > 0 ? zoneOf[k - W] : 0, b = row < W * (H - 1) ? zoneOf[k + W] : 0, c = x > 0 ? zoneOf[k - 1] : 0, d = x < W - 1 ? zoneOf[k + 1] : 0;
  if (!a && !b && !c && !d) continue;
  coast++; if (supply[k] === 0) { s0++; if (controller[k] !== 0) s0c++; }
}
console.log({ coast, supply0: s0, supply0ctl: s0c, layer: w.supplyBeach.size });
